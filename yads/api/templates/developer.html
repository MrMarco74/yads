Authentication

Pass your API key in the X-API-Key header on every request.

X-API-Key: yads_<your-key>

Keys are scoped to your tenant. read scope allows GET requests; write scope allows POST/PATCH/DELETE.

Code Examples

List targets

curl -H "X-API-Key: yads_YOUR_KEY" \
  {{ request.base_url }}api/v1/targets

Trigger a scan

curl -X POST \
  -H "X-API-Key: yads_YOUR_KEY" \
  -H "Content-Type: application/json" \
  -d '{"domain":"example.com","scan_types":["dns_scanner","ssl_scanner"]}' \
  {{ request.base_url }}api/v1/dast/scan

Get security findings

curl -H "X-API-Key: yads_YOUR_KEY" \
  "{{ request.base_url }}api/v1/findings?severity=critical"

Setup + list targets

import requests

BASE = "{{ request.base_url }}"
HEADERS = {"X-API-Key": "yads_YOUR_KEY"}

# List all targets
resp = requests.get(f"{BASE}api/v1/targets", headers=HEADERS)
targets = resp.json()
print(targets)

Trigger scan + poll result

import requests, time

BASE = "{{ request.base_url }}"
HEADERS = {"X-API-Key": "yads_YOUR_KEY"}

payload = {
    "domain": "example.com",
    "scan_types": ["dns_scanner", "ssl_scanner"]
}
resp = requests.post(
    f"{BASE}api/v1/dast/scan",
    json=payload, headers=HEADERS
)
task = resp.json()
print("Task queued:", task)

Fetch targets

const BASE = "{{ request.base_url }}";
const HEADERS = { "X-API-Key": "yads_YOUR_KEY" };

const resp = await fetch(`${BASE}api/v1/targets`, { headers: HEADERS });
const targets = await resp.json();
console.log(targets);

Trigger scan

const resp = await fetch(`${BASE}api/v1/dast/scan`, {
  method: "POST",
  headers: { ...HEADERS, "Content-Type": "application/json" },
  body: JSON.stringify({
    domain: "example.com",
    scan_types: ["dns_scanner", "ssl_scanner"]
  })
});
const result = await resp.json();
console.log(result);
